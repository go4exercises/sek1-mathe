\section*{3 · Musterlösung und Punkteraster}
\textbf{Allgemein:} Ganze Punkte je Zeile. Ein Fehler wird nur einmal abgezogen (Folgefehler). Gleichwertige Wege zählen voll.
In der Spalte P: \textbf{(E)}~= Ergebnispunkt, Ergebnis richtig \emph{und} Weg erkennbar; ohne Zeichen = die
Zeile bewertet den Weg, die Zeichnung oder die Begründung. Der ganze Test ist ohne Taschenrechner gestellt.
«Typische Fehler» nennt, welche Zeile 0 Punkte gibt, und die Punkte, die bleiben.
\textbf{Teilrichtige Zeilen:} Verlangt eine Zeile mehrere Angaben und stimmt nur ein Teil, gibt die Zeile 0 Punkte,
ausser die Zeile nennt ausdrücklich eine Aufteilung.
\textbf{Zahlen:} Dezimalzahl oder Bruch, wenn gleich ($4.5 = \tfrac{9}{2}$). Einheiten sind nicht verlangt.

\begin{lpaufgabe}{G1}{Werte}{3 P}
\begin{raster}
(a) Einsetzen & 1 & $-3 \cdot (-2) + 12 = 6 + 12$\\
(a) Ergebnis & 1\E & $y = 18$\\
(b) & 1\E & $-3x + 12 = -6$, $-3x = -18$, $x = 6$\\
\end{raster}
\fehler{(a) Minus mal Minus falsch ($-6 + 12 = 6$): Ergebniszeile (a) 0 $\to$ 2 von 3\,P.
(b) $-6$ für $x$ eingesetzt ($y = 30$): Zeile (b) 0.
(b) $-3x = -18$ richtig, dann $x = -6$: Zeile (b) 0.}
\end{lpaufgabe}

\begin{lpaufgabe}{G2}{Akku}{3 P}
\begin{raster}
(a) Gleichung & 1 & $y = -6x + 80$ (Anfangswert $b = 80$, Änderung $m = -6$ pro Stunde)\\
(b) & 1 & $-6 \cdot 5 + 80 = 50$, also $50\,\%$. Auch als Folgepunkt mit der eigenen Gleichung, wenn ein Ladestand zwischen $0\,\%$ und $100\,\%$ herauskommt.\\
(c) & 1\E & $-6x + 80 = 20$, $-6x = -60$, $x = 10$ Stunden. Direkt aus den Angaben lösbar ($80 - 20 = 60$, $60 : 6 = 10$), darum Ergebnispunkt.\\
\end{raster}
\fehler{$m$ und $b$ vertauscht ($y = 80x - 6$) oder Vorzeichen falsch ($y = 6x + 80$): Zeile (a) 0; (b) gibt keinen Folgepunkt ($394\,\%$ bzw.\ $110\,\%$ ist unmöglich) $\to$ höchstens 1 von 3\,P.
(c) $20 : 6$ gerechnet: Zeile (c) 0.
Mit $6\,\%$ des jeweiligen Stands gerechnet ($80 \cdot 0.94^5$): Das verlangt der Text nicht («Prozentpunkte»); Zeilen (a) und (b) 0.}
\end{lpaufgabe}

\begin{lpaufgabe}{G3}{Steigung}{3 P}
\begin{raster}
Ansatz & 1 & $m = \dfrac{4 - 1}{4 - (-2)}$: oben und unten dieselbe Reihenfolge ($B$ minus $A$ oder $A$ minus $B$)\\
Ergebnis & 1\E & $m = \dfrac{3}{6} = \dfrac{1}{2}$ (gleichwertig $0.5$)\\
Deutung & 1 & Die Gerade steigt (weil $m > 0$). Auch als Folgepunkt passend zum eigenen Vorzeichen.\\
\end{raster}
\fehler{Reihenfolge gemischt ($\tfrac{4 - 1}{-2 - 4} = -\tfrac{1}{2}$): Ansatz- und Ergebniszeile 0; «fällt» als Folgepunkt 1 $\to$ 1 von 3\,P.
Bruch umgedreht ($\tfrac{6}{3} = 2$): Ansatz- und Ergebniszeile 0; «steigt» 1 $\to$ 1 von 3\,P.
$4 - (-2) = 2$: Ergebniszeile 0 ($m = \tfrac{3}{2}$) $\to$ 2 von 3\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G4}{Zeichnen}{4 P}
\begin{center}
\gitter{-2}{9}{-4}{6}{\draw[lpblau,line width=1.2pt] (-2,13/3) -- (9,-3);
  \draw[lpgrau,line width=0.8pt] (0,3) -- (3,3) -- (3,1);
  \fill[lporange] (0,3) circle (2.4pt); \fill (3,1) circle (2.2pt); \fill[lpgruen] (6,-1) circle (2.4pt); \fill[lpgruen] (4.5,0) circle (2.4pt);}
\end{center}
\begin{raster}
Start & 1 & Punkt $(0 \mid 3)$ auf der $y$-Achse\\
Zweiter Punkt, Gerade & 1 & $3$ nach rechts, $2$ hinunter: $(3 \mid 1)$; Gerade durch beide Punkte (jeder richtige Weg zählt, auch eine Wertetabelle)\\
Punktprobe & 1\E & $-\tfrac{2}{3} \cdot 6 + 3 = -4 + 3 = -1$: ja, $P$ liegt darauf\\
Nullstelle & 1\E & $-\tfrac{2}{3}x + 3 = 0$, $\tfrac{2}{3}x = 3$, $2x = 9$, $x = 4.5$; Schnittpunkt $(4.5 \mid 0)$\\
\end{raster}
\fehler{Bei negativem $m$ hinauf gezeichnet (zweiter Punkt $(3 \mid 5)$): Zeile «Zweiter Punkt, Gerade» 0 $\to$ 3 von 4\,P.
Start bei $(0 \mid 0)$: Startzeile 0.
Punktprobe oder Nullstelle nur abgelesen, ohne Rechnung: diese Zeile 0.
Nullstelle mit Vorzeichenfehler ($x = -4.5$): Nullstellenzeile 0.}
\end{lpaufgabe}

\begin{lpaufgabe}{G5}{Ablesen}{3 P}
\begin{raster}
$b$ & 1 & $b = 1$ (Schnitt mit der $y$-Achse bei $(0 \mid 1)$)\\
$m$ & 1 & Von $(0 \mid 1)$ nach $(4 \mid -2)$: $4$ nach rechts, $3$ hinunter, $m = -\tfrac{3}{4}$\\
Gleichung & 1\E & $y = -\tfrac{3}{4}x + 1$ (gleichwertig $y = -0.75x + 1$)\\
\end{raster}
\fehler{Bruch umgedreht ($m = -\tfrac{4}{3}$): $m$-Zeile und Gleichungszeile 0 $\to$ 1 von 3\,P.
Vorzeichen vergessen ($m = \tfrac{3}{4}$): $m$-Zeile und Gleichungszeile 0 $\to$ 1 von 3\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G6}{Infusion}{3 P}
\begin{center}
\sachgitter{6}{8}{40}{100}{\draw[lpblau,line width=1.2pt] (0,8) -- (4,0);
  \fill[lporange] (0,8) circle (2.4pt); \fill[lpgruen] (4,0) circle (2.4pt);}
\end{center}
\begin{raster}
(a) & 1 & $V = -5t + 800$\\
(b) & 1\E & $-5t + 800 = 0$, $t = 160$ Minuten\\
(c) & 1 & Strecke von $(0 \mid 800)$ bis zum Ende aus (b) auf der $t$-Achse, nicht darunter weiter; das Ende als Folgepunkt mit dem eigenen Wert aus (b)\\
\end{raster}
\fehler{$V = 5t + 800$ oder $V = 800t - 5$: Zeile (a) 0; (b) gibt den Ergebnispunkt nur für $160$ mit erkennbarem Weg.
Gerade über das Ende hinaus ins Negative gezogen: Zeile (c) 0 (ein Beutel enthält nie weniger als $0$ ml).}
\end{lpaufgabe}

\begin{lpaufgabe}{G7}{Zwei Punkte}{3 P}
\begin{raster}
$m$ & 1 & $m = \dfrac{5 - (-1)}{6 - 2} = \dfrac{6}{4} = \dfrac{3}{2}$\\
$b$ & 1 & $P$ eingesetzt: $-1 = \tfrac{3}{2} \cdot 2 + b = 3 + b$, $b = -4$ (mit $Q$: $5 = 9 + b$). Auch als Folgepunkt mit dem eigenen $m$.\\
Gleichung, Probe & 1\E & $y = \tfrac{3}{2}x - 4$; Probe mit dem anderen Punkt, z.\,B.\ $Q$: $\tfrac{3}{2} \cdot 6 - 4 = 5$. Beides verlangt; eine Probe mit dem eingesetzten Punkt zählt nicht.\\
\end{raster}
\fehler{$b$ mit einem $y$-Wert verwechselt ($b = -1$ oder $b = 5$): $b$-Zeile und letzte Zeile 0 $\to$ 1 von 3\,P.
Bruch umgedreht ($m = \tfrac{2}{3}$, dann mit $P$: $b = -\tfrac{7}{3}$): $m$-Zeile 0, $b$-Zeile als Folgepunkt 1, letzte Zeile 0 $\to$ 1 von 3\,P.
Probe fehlt: letzte Zeile 0 $\to$ höchstens 2 von 3\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G8}{Neue Rate}{3 P}
\begin{center}
\sachgitter{6}{7}{50}{100}{\draw[lpblau,line width=1.2pt] (0,7) -- (1,4) -- (5,0);
  \fill[lporange] (1,4) circle (2.4pt); \fill[lpgruen] (5,0) circle (2.4pt);}
\end{center}
\begin{raster}
(a) & 1 & $700 - 6 \cdot 50 = 400$ ml\\
(b) & 1 & $400 : 2 = 200$ Minuten nach dem Umstellen, ab Beginn $50 + 200 = 250$ Minuten. Auch als Folgepunkt mit dem eigenen Wert aus (a).\\
(c) & 1 & Strecke $(0 \mid 700)$ bis $(50 \mid 400)$, dann flacher bis zum Ende aus (b): Knick bei $(50 \mid 400)$, kein Sprung. Das Ende als Folgepunkt mit dem eigenen Wert aus (b).\\
\end{raster}
\fehler{Nach dem Wechsel neu angefangen ($700 : 2 = 350$, leer nach $400$ Minuten, Graph springt auf $700$): Zeilen (b) und (c) 0 $\to$ 1 von 3\,P.
Wechsel übersehen ($700 : 6$, rund $117$ Minuten): Zeile (b) 0; (c) 0, wenn kein Knick gezeichnet ist.
Nur $200$ Minuten als Antwort (b), ohne die ersten $50$: Zeile (b) 0.}
\end{lpaufgabe}
