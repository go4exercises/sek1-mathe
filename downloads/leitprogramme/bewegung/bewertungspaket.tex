\section*{3 · Musterlösung und Punkteraster}
\textbf{Allgemein:} Ganze Punkte je Zeile. Ein Fehler wird nur einmal abgezogen (Folgefehler). Gleichwertige Wege zählen voll.
In der Spalte P: \textbf{(E)}~= Ergebnispunkt, Ergebnis richtig \emph{und} Weg erkennbar, direkt aus den Angaben; ohne Zeichen = die
Zeile bewertet den Weg, die Begründung oder den Antwortsatz. Der ganze Test ist ohne Taschenrechner gestellt.
«Typische Fehler» nennt, welche Zeile 0 Punkte gibt, und die Punkte, die bleiben.
\textbf{Teilrichtige Zeilen:} Verlangt eine Zeile mehrere Angaben und stimmt nur ein Teil, gibt die Zeile 0 Punkte,
ausser die Zeile nennt ausdrücklich eine Aufteilung.

\begin{lpaufgabe}{G1}{Streaming}{4 P}
\begin{raster}
Variable und Terme & 1 & $x$: Anzahl Filme pro Monat. A: $12 + 3x$, B: $30$ (Franken)\\
Gleichung & 1 & $12 + 3x = 30$ (oder eine Tabelle, die den Gleichstand zeigt)\\
Lösung & 1\E & $3x = 18$, $x = 6$: Bei $6$ Filmen kosten beide $30$ Fr.\\
Probe und Antwortsatz & 1 & $x = 7$: A $33$ Fr., B $30$ Fr. \emph{Ab $7$ Filmen im Monat ist B günstiger.}\\
\end{raster}
\fehler{Antwort «ab $6$ Filmen» oder nur «$x = 6$»: Zeile Probe und Antwortsatz 0 $\to$ 3 von 4\,P.
$12x + 3 = 30$ (Grundbetrag und Preis pro Film vertauscht): Zeilen Variable und Terme, Gleichung und Lösung 0; Probe und Antwortsatz als Folgepunkt, wenn sie zur eigenen Gleichung passen.}
\end{lpaufgabe}

\begin{lpaufgabe}{G2}{Zahlenrätsel}{3 P}
\begin{raster}
Gleichung & 1 & $n + (n + 2) + (n + 4) = 84$ (gerade Zahlen liegen $2$ auseinander)\\
Lösung & 1\E & $3n + 6 = 84$, $3n = 78$, $n = 26$: die Zahlen $26$, $28$, $30$\\
Probe & 1 & $26 + 28 + 30 = 84$, alle gerade\\
\end{raster}
\fehler{Abstand $1$ statt $2$ ($n + (n+1) + (n+2) = 84$, $n = 27$): Zeilen Gleichung und Lösung 0; Probe 1 nur, wenn sie zeigt, dass $27$ und $29$ ungerade sind.}
\end{lpaufgabe}

\begin{lpaufgabe}{G3}{Zugfahrt}{3 P}
\begin{raster}
(a) & 1\E & $1$ h $30$ min $= 1.5$ h, $v = 135 : 1.5 = 90$ km/h\\
(b) Zeit in h & 1 & $t = 60 : 90 = \tfrac{2}{3}$ h. Auch als Folgepunkt mit der eigenen Geschwindigkeit aus (a).\\
(b) in min & 1 & $\tfrac{2}{3} \cdot 60 = 40$ min\\
\end{raster}
\fehler{$1$ h $30$ min als $1.30$ h ($135 : 1.3 \approx 103.8$ km/h): Zeile (a) 0, (b) als Folgepunkte.
$90 : 60 = 1.5$ (Bruch umgekehrt): Zeile (b) Zeit in h 0.}
\end{lpaufgabe}

\begin{lpaufgabe}{G4}{Einheiten}{3 P}
\begin{raster}
(a) & 1\E & $54 : 3.6 = 15$ m/s\\
(b) & 1\E & $36 : 60 = 0.6$ h\\
(c) & 1\E & $4 \cdot 3.6 = 14.4$ km/h\\
\end{raster}
\fehler{(a) $54 \cdot 3.6$ oder (c) $4 : 3.6$ (Richtung verwechselt): diese Zeile 0. (b) $0.36$ h (Minuten als Hundertstel): Zeile (b) 0.}
\end{lpaufgabe}

\begin{lpaufgabe}{G5}{Bergwanderung}{4 P}
\begin{raster}
(a) Tabelle & 1 & $4$ km/h, $1.5$ h, $6$ km \;|\; Pause $0$ km/h, $0.5$ h, $0$ km \;|\; $2$ km/h, $2$ h, $4$ km\\
(b) & 1 & $1.5 + 0.5 + 2 = 4$ h. Auch als Folgepunkt mit den eigenen Zeiten aus (a).\\
(c) & 1\E & $10 : 4 = 2.5$ km/h (Gesamtweg durch Gesamtzeit, mit Pause)\\
(d) & 1 & Das Mittel stimmt nur bei gleich langen Zeiten. Der langsame Teil dauert länger (2 h statt 1.5 h) und zählt darum mehr; dazu kommt die Pause.\\
\end{raster}
\fehler{Pause vergessen ($10 : 3.5 \approx 2.9$ km/h): Zeilen (b) und (c) 0. Mittel $3$ km/h als Ergebnis von (c): Zeile (c) 0.}
\end{lpaufgabe}

\begin{lpaufgabe}{G6}{Plausibel?}{2 P}
\begin{raster}
(a) & 1 & $675$ h sind fast einen Monat: Für $45$ km braucht ein Velo nur wenige Stunden. (Oder: Mit $15$ km/h ist man in $3$ h schon $45$ km weit; die Einheit km·km/h ist keine Zeit.)\\
(b) & 1\E & $t = 45 : 15 = 3$ h\\
\end{raster}
\fehler{(a) nur «falsch gerechnet» ohne Grössenvergleich: Zeile (a) 0.}
\end{lpaufgabe}

\begin{lpaufgabe}{G7}{Begegnung}{3 P}
\begin{raster}
(a) & 1\E & aufeinander zu: $16 + 14 = 30$ km/h, $t = 45 : 30 = 1.5$ h\\
(b) & 1 & $16 \cdot 1.5 = 24$ km von Dorf A (Probe: Sara $14 \cdot 1.5 = 21$ km, $24 + 21 = 45$). Auch als Folgepunkt mit der eigenen Zeit aus (a).\\
(c) & 1 & z.\,B.: Beide fahren die ganze Zeit gleich schnell; sie starten genau gleichzeitig; keine Pausen, Ampeln oder Steigungen; die Strasse ist $45$ km lang.\\
\end{raster}
\fehler{$16 - 14 = 2$ km/h ($22.5$ h, gleiche Richtung gerechnet): Zeile (a) 0, (b) als Folgepunkt. Treffpunkt in der Mitte ($22.5$ km): Zeile (b) 0.}
\end{lpaufgabe}

\begin{lpaufgabe}{G8}{Rolltreppe}{3 P}
\begin{raster}
(a) & 1\E & $1.2 + 0.8 = 2$ m/s, $24 : 2 = 12$ s\\
(b) & 1\E & $1.2 - 0.8 = 0.4$ m/s, $24 : 0.4 = 60$ s\\
(c) & 1\E & $24 : 0.8 = 30$ s\\
\end{raster}
\fehler{(b) mit $2$ m/s gerechnet ($12$ s, addiert statt subtrahiert): Zeile (b) 0 $\to$ 2 von 3\,P.}
\end{lpaufgabe}
