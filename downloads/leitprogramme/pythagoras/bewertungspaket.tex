\section*{3 · Musterlösung und Punkteraster}
\textbf{Allgemein:} Ganze Punkte je Zeile. Ein Fehler wird nur einmal abgezogen (Folgefehler). Gleichwertige Wege zählen voll.
In der Spalte P: \textbf{(E)}~= Ergebnispunkt, Ergebnis richtig \emph{und} Weg erkennbar, direkt aus den Angaben; ohne Zeichen = die
Zeile bewertet den Weg, die Skizze oder die Begründung. Der ganze Test ist ohne Taschenrechner gestellt, alle Wurzeln gehen auf.
\textbf{Einheiten:} Fehlt nur die Einheit, gibt es keinen Abzug, ausser die Zeile verlangt eine Umrechnung (G8).
«Typische Fehler» nennt, welche Zeile 0 Punkte gibt, und die Punkte, die bleiben.
\textbf{Teilrichtige Zeilen:} Verlangt eine Zeile mehrere Angaben und stimmt nur ein Teil, gibt die Zeile 0 Punkte,
ausser die Zeile nennt ausdrücklich eine Aufteilung.

\begin{lpaufgabe}{G1}{Giebelwand}{3 P}
\begin{raster}
Zerlegen & 1 & Rechteck $10 \cdot 6 = 60$ m$^2$\\
Dreieck & 1 & $\dfrac{10 \cdot 4}{2} = 20$ m$^2$ (Höhe $4$, nicht die schräge Seite $6.4$)\\
Summe & 1 & $60 + 20 = 80$ m$^2$. Auch als Folgepunkt aus den eigenen Teilflächen.\\
\end{raster}
\fehler{Dreieck mit der schrägen Seite ($\tfrac{10 \cdot 6.4}{2} = 32$, total $92$): Zeile Dreieck 0 $\to$ 2 von 3\,P.
Dreieck nicht halbiert ($40$, total $100$): Zeile Dreieck 0 $\to$ 2 von 3\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G2}{Trapez und Parallelogramm}{3 P}
\begin{raster}
(a) Ansatz & 1 & $\dfrac{14 + 8}{2} \cdot 5$\\
(a) Ergebnis & 1\E & $11 \cdot 5 = 55$ cm$^2$\\
(b) & 1\E & $9 \cdot 6 = 54$ cm$^2$; die schräge Seite $7.5$ cm wird nicht gebraucht\\
\end{raster}
\fehler{(a) $(14 + 8) \cdot 5 = 110$ (nicht halbiert): Zeilen (a) 0 $\to$ 1 von 3\,P. (b) $9 \cdot 7.5 = 67.5$ (schräge Seite): Zeile (b) 0 $\to$ 2 von 3\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G3}{Hypotenuse}{3 P}
\begin{raster}
(a) Ansatz & 1 & $c^2 = 16^2 + 30^2 = 256 + 900 = 1156$\\
(a) Ergebnis & 1\E & $c = \sqrt{1156} = 34$ cm\\
(b) & 1 & $10^2 + 24^2 = 100 + 576 = 676$, aber $25^2 = 625$: nicht rechtwinklig (Rechnung und Schluss verlangt)\\
\end{raster}
\fehler{(a) $16 + 30 = 46$ (Seiten addiert): beide Zeilen (a) 0 $\to$ 1 von 3\,P. (a) Wurzel vergessen ($c = 1156$): Zeile (a) Ergebnis 0 $\to$ 2 von 3\,P.
(b) «ja, weil $25$ die längste Seite ist» ohne Rechnung: Zeile (b) 0.}
\end{lpaufgabe}

\begin{lpaufgabe}{G4}{Leiter}{3 P}
\begin{raster}
Skizze & 1 & rechtwinkliges Dreieck aus Wand, Boden und Leiter; die Leiter ($6.5$ m) ist die Hypotenuse, der rechte Winkel zwischen Wand und Boden\\
Ansatz & 1 & $h^2 = 6.5^2 - 2.5^2 = 42.25 - 6.25 = 36$\\
Ergebnis & 1\E & $h = 6$ m\\
\end{raster}
\fehler{$h^2 = 6.5^2 + 2.5^2$ (Kathete mit plus): Zeilen Ansatz und Ergebnis 0 $\to$ 1 von 3\,P. $6.5 - 2.5 = 4$ (Längen subtrahiert): Zeilen Ansatz und Ergebnis 0.}
\end{lpaufgabe}

\begin{lpaufgabe}{G5}{Gleichschenkliges Dreieck}{4 P}
\begin{raster}
(a) Skizze & 1 & Höhe von der Spitze senkrecht auf die Basis; Hilfsdreieck mit halber Basis $21$ cm, Höhe $h$ und Schenkel $29$ cm als Hypotenuse\\
(b) Ansatz & 1 & $h^2 = 29^2 - 21^2 = 841 - 441 = 400$\\
(b) Ergebnis & 1\E & $h = 20$ cm\\
(c) & 1 & $\dfrac{42 \cdot 20}{2} = 420$ cm$^2$. Auch als Folgepunkt mit der eigenen Höhe.\\
\end{raster}
\fehler{Ganze Basis ins Hilfsdreieck ($29^2 - 42^2 < 0$ oder $42$ als Kathete): Zeilen (a) und (b) Ansatz 0; meist kein Ergebnis $\to$ höchstens 1 von 4\,P.
(c) nicht halbiert ($840$): Zeile (c) 0 $\to$ 3 von 4\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G6}{Wasser im Rohr}{3 P}
\begin{raster}
Hilfsdreieck & 1 & Radius $15$ cm als Hypotenuse (vom Mittelpunkt zum Rand des Wasserspiegels), halbe Breite $12$ cm, Abstand $d$ senkrecht zum Spiegel\\
Abstand & 1 & $d = \sqrt{15^2 - 12^2} = \sqrt{81} = 9$ cm. Auch als Folgepunkt mit dem eigenen Hilfsdreieck.\\
Tiefe & 1\E & $15 - 9 = 6$ cm\\
\end{raster}
\fehler{Durchmesser $30$ als Hypotenuse ($\sqrt{900 - 144}$): Zeilen Hilfsdreieck und Abstand 0 $\to$ höchstens 1 von 3\,P.
Ganze Breite $24$ statt $12$: Zeile Hilfsdreieck 0. Tiefe $15 + 9 = 24$ (Spiegel über der Mitte angenommen): Zeile Tiefe 0 $\to$ 2 von 3\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G7}{Regal}{3 P}
\begin{raster}
(a) & 1\E & $k = \dfrac{30}{90} = \dfrac{1}{3}$\\
(b) & 1 & grosses Dreieck $\dfrac{90 \cdot 60}{2} = 2700$ cm$^2$; ausgesägt $2700 \cdot \left(\tfrac{1}{3}\right)^2 = 300$ cm$^2$ (oder Katheten $30$ und $20$: $\tfrac{30 \cdot 20}{2} = 300$)\\
(c) & 1 & $2700 - 300 = 2400$ cm$^2$. Auch als Folgepunkt aus den eigenen Flächen.\\
\end{raster}
\fehler{(b) $2700 \cdot \tfrac{1}{3} = 900$ (Flächenfaktor $k$ statt $k^2$): Zeile (b) 0, (c) als Folgepunkt ($1800$) $\to$ 2 von 3\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G8}{Modell}{3 P}
\begin{raster}
(a) & 1 & Länge \emph{und} Breite werden mal $50$ gerechnet; die Fläche wächst darum mit $50^2 = 2500$, nicht mit $50$.\\
(b) Fläche & 1 & $12 \cdot 2500 = 30\,000$ cm$^2$\\
(b) Umrechnung & 1 & $30\,000$ cm$^2$ $= 3$ m$^2$ (denn $1$ m$^2 = 10\,000$ cm$^2$). Auch als Folgepunkt aus der eigenen Fläche.\\
\end{raster}
\fehler{(b) $600$ cm$^2$ übernommen: Zeile (b) Fläche 0, Umrechnung als Folgepunkt ($0.06$ m$^2$) $\to$ 1 oder 2 von 3\,P.
$30\,000$ cm$^2$ $= 300$ m$^2$ (mit $100$ umgerechnet): Zeile Umrechnung 0 $\to$ 2 von 3\,P.}
\end{lpaufgabe}
